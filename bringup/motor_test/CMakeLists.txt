cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(motor_bench)
target_sources(app PRIVATE src/main.c src/bench_control.c src/bench_encoders.c src/velocity_control.c)
# Startup and HOLD diagnostic profiles are fixed per image; PID controls duty at runtime.
set(MOTOR_TEST_DUTY "60" CACHE STRING "Startup-test duty percent, 1..100")
if(NOT MOTOR_TEST_DUTY MATCHES "^[1-9][0-9]?$|^100$")
  message(FATAL_ERROR "MOTOR_TEST_DUTY must be an integer from 1 to 100")
endif()
target_compile_definitions(app PRIVATE BENCH_DUTY_PERCENT=${MOTOR_TEST_DUTY}U)
set(MOTOR_HOLD_DUTY "55" CACHE STRING "Holding-test duty percent, 1..60")
if(NOT MOTOR_HOLD_DUTY MATCHES "^[1-9]$|^[1-5][0-9]$|^60$")
  message(FATAL_ERROR "MOTOR_HOLD_DUTY must be an integer from 1 to 60")
endif()
set(MOTOR_HOLD_MS "2000" CACHE STRING "Holding-test limit, 2000 or 4000 ms")
if(NOT MOTOR_HOLD_MS MATCHES "^(2000|4000)$")
  message(FATAL_ERROR "MOTOR_HOLD_MS must be 2000 or 4000")
endif()
target_compile_definitions(app PRIVATE BENCH_HOLD_PERCENT=${MOTOR_HOLD_DUTY}U
                                       BENCH_HOLD_MS=${MOTOR_HOLD_MS})
